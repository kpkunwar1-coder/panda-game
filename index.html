// --- LIVES SYSTEM LOGIC --- const MAX_LIVES = 5; const REFILL_TIME = 30 * 60 * 1000; // 30 Minutes in milliseconds let gameState = { lives: parseInt(localStorage.getItem('pandaLives')) ?? MAX_LIVES, lastLoss: parseInt(localStorage.getItem('pandaLastLoss')) || Date.now() }; function updateLives() { const now = Date.now(); const timePassed = now - gameState.lastLoss; // Calculate how many hearts regenerated if (gameState.lives < MAX_LIVES) { const regenerated = Math.floor(timePassed / REFILL_TIME); if (regenerated > 0) { gameState.lives = Math.min(MAX_LIVES, gameState.lives + regenerated); gameState.lastLoss = now; // Reset timer for the next heart saveState(); } } renderHearts(); } function renderHearts() { const heartContainer = document.getElementById('heart-display'); heartContainer.innerHTML = '❤️'.repeat(gameState.lives) + '🖤'.repeat(MAX_LIVES - gameState.lives); // Show timer if lives are missing const timer = document.getElementById('heart-timer'); if (gameState.lives < MAX_LIVES) { const remaining = REFILL_TIME - (Date.now() - gameState.lastLoss); const mins = Math.floor(remaining / 60000); const secs = Math.floor((remaining % 60000) / 1000); timer.innerText = `Next in: ${mins}:${secs < 10 ? '0' : ''}${secs}`; } else { timer.innerText = "Full"; } } function loseLife() { if (gameState.lives > 0) { if (gameState.lives === MAX_LIVES) gameState.lastLoss = Date.now(); gameState.lives--; saveState(); renderHearts(); if(window.navigator.vibrate) window.navigator.vibrate([100, 50, 100]); // "Oops" vibration } if (gameState.lives === 0) { alert("Out of lives! Wait for regeneration."); } } function saveState() { localStorage.setItem('pandaLives', gameState.lives); localStorage.setItem('pandaLastLoss', gameState.lastLoss); } // Update hearts every second for the timer setInterval(updateLives, 1000);
